namespace Alarmlist.Compiler
{
    public class AlarmCompiler
    {
        public Alarmlist.Analysis.AnalysisResult AnalyzeSnapshots(IEnumerable<Alarmlist.Text.AlmxTextSnapshot> snapshots)
        {
            var sources = new List<Alarmlist.Analysis.AnalyzedSource>();
            foreach (var snapshot in snapshots)
                sources.Add(new Alarmlist.Analysis.AnalyzedSource { FilePath = snapshot.FilePath, Version = snapshot.Version });
            return new Alarmlist.Analysis.AnalysisResult { Sources = sources };
        }
        // OPTIONAL_PREVIEW_METHOD
    }
